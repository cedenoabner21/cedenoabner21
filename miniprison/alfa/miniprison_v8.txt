function pintarCelda(col, row, tipo) {

  celda.tipo = tipo;
  gridDirty = true;
  return true;
}

function pintarLinea(c0, r0, c1, r1) {
  // BUG 10: no pintar si la celda destino es inválida
  if (c1 < 0 || r1 < 0) return;
  if (c0 < 0 || r0 < 0) {
    pintarCelda(c1, r1, tipoSeleccionado);
    return;
  }
  let dc = Math.abs(c1 - c0);
  let dr = Math.abs(r1 - r0);
  let sc = c0 < c1 ? 1 : -1;
  let sr = r0 < r1 ? 1 : -1;
  let err = dc - dr;
  let c = c0, r = r0;
  while (true) {
    pintarCelda(c, r, tipoSeleccionado);
    if (c === c1 && r === r1) break;
    const e2 = 2 * err;
    if (e2 > -dr) { err -= dr; c += sc; }
    if (e2 <  dc) { err += dc; r += sr; }
  }
}

/* =========================================================
   MEJORA 1: BFS con `head` en lugar de `shift()`
   `shift()` es O(n) por cada extracción; con `head` que avanza
   el coste total es O(n) para toda la búsqueda.
   ========================================================= */
function bfsCeldaMasCercana(startCol, startRow, tipoBuscado) {
  if (startCol < 0 || startRow < 0 || startCol >= COLS || startRow >= ROWS) {
    return null;
  }
  const visitado = new Uint8Array(COLS * ROWS);
  const cola = [{ col: startCol, row: startRow }];
  visitado[startRow * COLS + startCol] = 1;
  let head = 0;

  const dirs = [[0,-1],[0,1],[-1,0],[1,0]];

  while (head < cola.length) {
    const nodo = cola[head++];
    const col = nodo.col;
    const row = nodo.row;

    if (grid[row][col].tipo === tipoBuscado) {
      return { col, row };
    }

    for (let d = 0; d < 4; d++) {
      const dc = dirs[d][0];
      const dr = dirs[d][1];
      const nc = col + dc;
      const nr = row + dr;
      if (nc < 0 || nr < 0 || nc >= COLS || nr >= ROWS) continue;
      const idx = nr * COLS + nc;
      if (visitado[idx]) continue;
      if (grid[nr][nc].tipo === "muro") continue;
      visitado[idx] = 1;
      cola.push({ col: nc, row: nr });
    }
  }
  return null;
}

function celdaDePunto(x, y) {
  return {
    col: Math.max(0, Math.min(COLS - 1, Math.floor(x / TILE))),
    row: Math.max(0, Math.min(ROWS - 1, Math.floor(y / TILE)))
  };
}

/* =========================================================
   LÓGICA DE NECESIDADES
   ========================================================= */
function asignarTargetATipo(npc, tipoCelda, motivo) {
  const p = celdaDePunto(npc.x, npc.y);
  const destino = bfsCeldaMasCercana(p.col, p.row, tipoCelda);
  if (destino) {
    npc.targetX = destino.col * TILE + TILE / 2;
    npc.targetY = destino.row * TILE + TILE / 2;
    npc.objetivo = motivo;
    return true;
  }
  return false;
}

/* MEJORA 2: usa indiceCeldasPorTipo en lugar de recorrer todo el grid */
function asignarTargetAleatorioDeTipo(npc, tipoCelda, motivo) {
  const arr = indiceCeldasPorTipo.get(tipoCelda);
  if (!arr || arr.length === 0) return false;
  const dest = arr[Math.floor(Math.random() * arr.length)];
  npc.targetX = dest.col * TILE + TILE / 2;
  npc.targetY = dest.row * TILE + TILE / 2;
  npc.objetivo = motivo;
  return true;
}

/* MEJORA 5: sqrt en lugar de hypot en punto caliente */
function enDestino(npc) {
  const dx = npc.targetX - npc.x;
  const dy = npc.targetY - npc.y;
  return Math.sqrt(dx * dx + dy * dy) < 4;
}

function resolverLlegada(npc) {
  if (npc.tipo !== "preso") return;
  if (npc.objetivo === "comedor") {
    npc.hambre = 100;
  } else if (npc.objetivo === "ducha") {
    npc.higiene = 100;
  } else if (npc.objetivo === "celda") {
    npc.libertad = Math.min(100, npc.libertad + 5);
  }
  npc.objetivo = null;
  npc.targetX = npc.x;
  npc.targetY = npc.y;
}

function tickPreso(preso) {
  preso.hambre   = Math.max(0, preso.hambre   - 2);
  preso.higiene  = Math.max(0, preso.higiene  - 1);
  preso.libertad = Math.max(0, preso.libertad - 0.5);

  if (preso.libertad <= 0) {
    preso.estado = "motin";
  } else if (preso.libertad < 20) {
    preso.estado = "inquieto";
  } else {
    preso.estado = "normal";
  }

  if (preso.estado === "inquieto" || preso.estado === "motin") {
    preso.speed = preso.speedBase * SPEED_INQUIETO;
  } else {
    preso.speed = preso.speedBase;
  }

  if (preso.estado === "motin") {
    if (enDestino(preso) || preso.objetivo === null) {
      if (!asignarTargetAleatorioDeTipo(preso, "patio", "patio")) {
        asignarTargetAleatorioDeTipo(preso, "vacio", "patio");
      }
    }
    return;
  }

  if (preso.objetivo && enDestino(preso)) {
    resolverLlegada(preso);
  }

  if (!preso.objetivo) {
    if (preso.hambre < 30) {
      if (!asignarTargetATipo(preso, "comedor", "comedor")) {
        asignarTargetAleatorioDeTipo(preso, "vacio", "patio");
      }
    } else if (preso.higiene < 30) {
      if (!asignarTargetATipo(preso, "ducha", "ducha")) {
        asignarTargetAleatorioDeTipo(preso, "vacio", "patio");
      }
    } else {
      if (!asignarTargetAleatorioDeTipo(preso, "patio", "patio")) {
        asignarTargetAleatorioDeTipo(preso, "vacio", "patio");
      }
    }
  }
}

const DIST_DETECTAR_MOTIN = 100;
const DIST_CALMAR        = 20;

/* BUG 9: recorre solo la lista pre-filtrada por tick */
function presoEnMotinMasCercano(guardia) {
  let mejor = null;
  let mejorDist = Infinity;
  const gx = guardia.x;
  const gy = guardia.y;
  for (let i = 0; i < presosEnMotin.length; i++) {
    const n = presosEnMotin[i];
    const dx = n.x - gx;
    const dy = n.y - gy;
    // MEJORA 5: sqrt en lugar de hypot
    const d = Math.sqrt(dx * dx + dy * dy);
    if (d < mejorDist) {
      mejorDist = d;
      mejor = n;
    }
  }
  return mejor;
}

function patrullar(guardia) {
  for (let i = 0; i < 50; i++) {
    const col = Math.floor(Math.random() * COLS);
    const row = Math.floor(Math.random() * ROWS);
    if (celdaEsTransitable(col, row)) {
      guardia.targetX = col * TILE + TILE / 2;
      guardia.targetY = row * TILE + TILE / 2;
      guardia.objetivo = "patrulla";
      return;
    }
  }
}

function tickGuardia(guardia) {
  guardia.detectoMotin = false;

  const motin = presoEnMotinMasCercano(guardia);
  if (motin) {
    const dx = motin.x - guardia.x;
    const dy = motin.y - guardia.y;
    // MEJORA 5: sqrt en lugar de hypot
    const d = Math.sqrt(dx * dx + dy * dy);
    if (d < DIST_DETECTAR_MOTIN) {
      guardia.detectoMotin = true;
      guardia.objetivo = "perseguir";
      guardia.presoObjetivo = motin;
      guardia.targetX = motin.x;
      guardia.targetY = motin.y;

      if (d < DIST_CALMAR) {
        motin.estado = "normal";
        motin.libertad = 50;
        motin.speed = motin.speedBase;
        motin.objetivo = null;
        motin.targetX = motin.x;
        motin.targetY = motin.y;
        guardia.objetivo = "patrulla";
        guardia.presoObjetivo = null;
      }
      return;
    }
  }

  if (guardia.objetivo !== "patrulla" || enDestino(guardia)) {
    guardia.presoObjetivo = null;
    patrullar(guardia);
  }
}

}

/* MEJORA 3: caché de hexToRgba con límite de 64 entradas */
const cacheRgba = new Map();

function hexToRgba(hex, alpha) {
  const clave = hex + "_" + alpha;
  let val = cacheRgba.get(clave);
  if (val !== undefined) return val;
  if (cacheRgba.size >= 64) cacheRgba.clear();
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  val = "rgba(" + r + "," + g + "," + b + "," + alpha + ")";
  cacheRgba.set(clave, val);
  return val;
}

function dibujarNPC(npc, offsetX, offsetY, tiempoMs) {
  const sx = npc.x + offsetX;
  const sy = npc.y + offsetY;

  // Sombra suave
  ctx.beginPath();
  ctx.ellipse(sx, sy + NPC_RADIUS * 0.9, NPC_RADIUS * 1.2, NPC_RADIUS * 0.5, 0, 0, Math.PI * 2);
  ctx.fillStyle = "rgba(0,0,0,0.2)";
  ctx.fill();

  let colorCono = npc.color;
  if (npc.tipo === "preso") {
    if (npc.estado === "inquieto") colorCono = "#ff8800";
    else if (npc.estado === "motin") colorCono = "#ff0000";
  } else if (npc.tipo === "guardia" && npc.detectoMotin) {
    colorCono = "#ff6666";
  }

  const a1 = npc.angle - VISION_FOV / 2;
  const a2 = npc.angle + VISION_FOV / 2;
  ctx.beginPath();
  ctx.moveTo(sx, sy);
  ctx.lineTo(sx + Math.cos(a1) * VISION_RADIUS,
             sy + Math.sin(a1) * VISION_RADIUS);
  ctx.lineTo(sx + Math.cos(a2) * VISION_RADIUS,
             sy + Math.sin(a2) * VISION_RADIUS);
  ctx.closePath();
  ctx.fillStyle = hexToRgba(colorCono, VISION_ALPHA);
  ctx.fill();

  ctx.beginPath();
  ctx.arc(sx, sy, NPC_RADIUS, 0, Math.PI * 2);

  if (npc.tipo === "preso" && npc.estado === "motin") {
    const pulso = 0.5 + 0.5 * Math.sin(tiempoMs * 0.008);
    const r = 255;
    const g = Math.floor(60 + 120 * pulso);
    const b = Math.floor(60 + 40 * pulso);
    ctx.fillStyle = "rgb(" + r + "," + g + "," + b + ")";
  } else {
    ctx.fillStyle = npc.color;
  }
  ctx.fill();

  ctx.lineWidth = 2;
  ctx.strokeStyle = "#ffffff";
  ctx.stroke();
}

/* =========================================================
   POBLAR EL MUNDO
   ========================================================= */
function posAleatoriaTransitable() {
  for (let i = 0; i < 50; i++) {
    const col = Math.floor(Math.random() * COLS);
    const row = Math.floor(Math.random() * ROWS);
    if (celdaEsTransitable(col, row)) {
      return {
        x: col * TILE + TILE / 2,
        y: row * TILE + TILE / 2
      };
    }
  }
  return { x: TILE / 2, y: TILE / 2 };
}

function poblarMundo() {
  for (let i = 0; i < 5; i++) {
    const p = posAleatoriaTransitable();
    npcs.push(crearPreso(p.x, p.y));
  }
  for (let i = 0; i < 2; i++) {
    const p = posAleatoriaTransitable();
    npcs.push(crearGuardia(p.x, p.y));
  }
  const p = posAleatoriaTransitable();
  npcs.push(crearObrero(p.x, p.y));
}

/* =========================================================
   ACTUALIZACIÓN DE MOVIMIENTO
   ========================================================= */
function lerpAngle(a, b, t) {
  let diff = ((b - a + Math.PI * 3) % (Math.PI * 2)) - Math.PI;
  return a + diff * t;
}

/* MEJORA 5: sqrt en lugar de hypot */
function actualizarMovimiento(npc, dt) {
  const dx = npc.targetX - npc.x;
  const dy = npc.targetY - npc.y;
  const dist = Math.sqrt(dx * dx + dy * dy);

  if (dist < 2) return;

  const dirDeseada = Math.atan2(dy, dx);
  const t = Math.min(1, dt * 8);
  npc.angle = lerpAngle(npc.angle, dirDeseada, t);

  const step = npc.speed * dt;
  const nx = npc.x + Math.cos(dirDeseada) * Math.min(step, dist);
  const ny = npc.y + Math.sin(dirDeseada) * Math.min(step, dist);

  if (puntoEnMuro(nx, ny)) return;

  npc.x = nx;
  npc.y = ny;
}

const INGRESO_POR_PRESO = 20;
const GASTO_POR_GUARDIA = 15;
const GASTO_POR_OBRERO  = 10;

function contarTipo(tipo) {
  let n = 0;
  for (let i = 0; i < npcs.length; i++) {
    if (npcs[i].tipo === tipo) n++;
  }
  return n;
}

/* MEJORA 2: usa el índice en lugar de recorrer todo el grid */
function celdasLibresDeCelda() {
  const arr = indiceCeldasPorTipo.get("celda");
  const total = arr ? arr.length : 0;
  const ocupadas = contarTipo("preso");
  return Math.max(0, total - ocupadas);
}

function cerrarDia() {
  // BUG 3: evitar doble cierre del mismo día
  if (estadoGlobal.dia === estadoGlobal.ultimoDiaCerrado) return;

  const presos   = contarTipo("preso");
  const guardias = contarTipo("guardia");
  const obreros  = contarTipo("obrero");

  const ingreso = presos * INGRESO_POR_PRESO;
  const gasto   = guardias * GASTO_POR_GUARDIA + obreros * GASTO_POR_OBRERO;

  estadoGlobal.dinero += ingreso;
  estadoGlobal.dinero -= gasto;

  if (estadoGlobal.dinero < 0) {
    estadoGlobal.dinero = 0;
    finalizarPartida("derrota", "Te quedaste sin dinero");
    estadoGlobal.ultimoDiaCerrado = estadoGlobal.dia;
    return;
  }

  estadoGlobal.dia++;
  estadoGlobal.tickDelDia = 0;

  // Llegada de presos: 1 nuevo preso si hay celdas libres de tipo "celda"
  if (estadoGlobal.dia <= DIA_VICTORIA && celdasLibresDeCelda() > 0) {
    const p = posAleatoriaTransitable();
    npcs.push(crearPreso(p.x, p.y));
  }

  // Chequeo de victoria / derrota por objetivo
  if (estadoGlobal.dia > DIA_VICTORIA) {
    const presosFinal = contarTipo("preso");
    if (presosFinal >= PRESOS_PARA_GANAR) {
      finalizarPartida("victoria", "¡Has construido una prisión exitosa!");
    } else if (presosFinal < PRESOS_MINIMOS) {
      finalizarPartida("derrota", "Muy pocos presos al final del día " + DIA_VICTORIA);
    } else {
      finalizarPartida("derrota", "No llegaste a " + PRESOS_PARA_GANAR + " presos");
    }
  }

  // BUG 3: marcar el día como cerrado
  estadoGlobal.ultimoDiaCerrado = estadoGlobal.dia - 1;
}

function finalizarPartida(resultado, motivo) {
  if (estadoGlobal.finDePartida) return;
  estadoGlobal.finDePartida = true;
  estadoGlobal.pausado = true;
  estadoGlobal.resultado = resultado;
  estadoGlobal.motivoFin = motivo;
}

 * tick ya no están en estado "motin" al llegar aquí, así que no cuentan.
 * Esto es intencional: un motín debe durar al menos un tick completo.

function comprobarMotines() {
  let motines = 0;
  for (let i = 0; i < npcs.length; i++) {
    const n = npcs[i];
    if (n.tipo === "preso" && n.estado === "motin") motines++;
  }
  estadoGlobal.motinesActuales = motines;
  if (motines >= MOTINES_MAX && !estadoGlobal.finDePartida) {
    finalizarPartida("derrota", "Motín general: " + motines + " presos en motín");
  }
}

let lastTick = performance.now();

function tick() {
  // BUG 11: helper unificado
  if (!puedeSimular()) return;

  estadoGlobal.tickDelDia++;
  if (estadoGlobal.tickDelDia >= TICKS_POR_DIA) {
    cerrarDia();
    if (estadoGlobal.finDePartida) return;
  }

  // BUG 9: reconstruir lista de presos en motín UNA vez por tick
  presosEnMotin.length = 0;
  for (let i = 0; i < npcs.length; i++) {
    const n = npcs[i];
    if (n.tipo === "preso" && n.estado === "motin") presosEnMotin.push(n);
  }

  for (let i = 0; i < npcs.length; i++) {
    const npc = npcs[i];
    if (npc.tipo === "preso") tickPreso(npc);
    else if (npc.tipo === "guardia") tickGuardia(npc);
    else if (npc.tipo === "obrero") tickObrero(npc);
  }

  comprobarMotines();
}

function faseDelDia() {
  // BUG 6: guarda contra TICKS_POR_DIA <= 0
  if (TICKS_POR_DIA <= 0) {
    return { nombre: "Noche", tinte: "rgba(10,10,40,0.22)" };
  }
  const p = estadoGlobal.tickDelDia / TICKS_POR_DIA;

  let a, b, t;
  if (p < 0.25)      { a = FASE_AMANECER; b = FASE_MEDIODIA; t = p / 0.25; }
  else if (p < 0.5)  { a = FASE_MEDIODIA; b = FASE_TARDE;    t = (p - 0.25) / 0.25; }
  else if (p < 0.75) { a = FASE_TARDE;    b = FASE_NOCHE;    t = (p - 0.5) / 0.25; }
  else               { a = FASE_NOCHE;    b = FASE_AMANECER; t = (p - 0.75) / 0.25; }

  const r = Math.round(a.r + (b.r - a.r) * t);
  const g = Math.round(a.g + (b.g - a.g) * t);
  const bl = Math.round(a.b + (b.b - a.b) * t);
  const alpha = a.a + (b.a - a.a) * t;

  let nombre;
  if (p < 0.20)      nombre = "Amanecer";
  else if (p < 0.45) nombre = "Mediodía";
  else if (p < 0.70) nombre = "Atardecer";
  else               nombre = "Noche";

  return {
    nombre,
    tinte: "rgba(" + r + "," + g + "," + bl + "," + alpha.toFixed(3) + ")"
  };
}

/* BUG 7: fallback de padStart con concatenación manual */
function dosDigitos(n) {
  return n < 10 ? "0" + n : "" + n;
}

function horaDelDia() {
  // BUG 6: guarda contra TICKS_POR_DIA <= 0
  if (TICKS_POR_DIA <= 0) return "00:00";
  const minutos = Math.floor(
    (estadoGlobal.tickDelDia / TICKS_POR_DIA) * MINUTOS_POR_DIA
  );
  const hh = Math.floor(minutos / 60) % 24;
  const mm = minutos % 60;
  return dosDigitos(hh) + ":" + dosDigitos(mm);
}

/* =========================================================
   BUCLE PRINCIPAL
   ========================================================= */
let lastFrameTime = performance.now();

function loop(now) {
  const dtMs = now - lastFrameTime;
  lastFrameTime = now;
  const dt = Math.min(dtMs / 1000, 0.05);

  // --- Ticks de simulación ---
  let ticksRun = 0;
  while (now - lastTick >= TICK_MS && ticksRun < 5) {
    lastTick += TICK_MS;
    tick();
    ticksRun++;
  }
  if (now - lastTick >= TICK_MS) lastTick = now;

  // --- Movimiento de NPCs (BUG 11: helper unificado) ---
  if (puedeSimular()) {
    for (let i = 0; i < npcs.length; i++) {
      actualizarMovimiento(npcs[i], dt);
    }
  }

  // --- FPS ---
  fpsAccum += dtMs;
  fpsFrames++;
  if (fpsAccum >= 500) {
    fpsValue = Math.round((fpsFrames * 1000) / fpsAccum);
    fpsAccum = 0;
    fpsFrames = 0;
  }

  // --- Render ---
  draw(now);

  requestAnimationFrame(loop);
}

poblarMundo();
requestAnimationFrame(loop);
