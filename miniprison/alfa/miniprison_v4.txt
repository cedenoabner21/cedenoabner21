<script>
"use strict";

/* =========================================================
   CONFIGURACIÓN GLOBAL
   ========================================================= */
const TILE = 40;
const COLS = 20;
const ROWS = 15;
const GRID_W = TILE * COLS;
const GRID_H = TILE * ROWS;

const COLOR_BG = "#1a1a2e";
const COLOR_GRID = "rgba(255,255,255,0.08)";

const TICK_MS = 1000;

// Parámetros de los NPCs
const NPC_RADIUS = 10;
const VISION_RADIUS = 80;
const VISION_FOV = Math.PI / 3;
const VISION_ALPHA = 0.15;

const NPC_COLORS = {
  preso:   "#ff4444",
  guardia: "#4488ff",
  obrero:  "#ffcc44"
};

const NPC_SPEEDS = {
  preso:   60,
  guardia: 90,
  obrero:  70
};

/* =========================================================
   TIPOS DE CELDA Y COLORES
   ========================================================= */
// Mapa de tipos → color de relleno de la celda
const TIPOS_CELDA = {
  vacio:   { color: null },
  muro:    { color: "#555555" },
  celda:   { color: "#333355" },
  comedor: { color: "#335533" },
  ducha:   { color: "#334455" },
  patio:   { color: "#555533" }
};

// Orden de las teclas 1-6 → tipo
const TECLAS_TIPO = ["vacio", "muro", "celda", "comedor", "ducha", "patio"];

/* =========================================================
   CANVAS A PANTALLA COMPLETA (con soporte HiDPI)
   ========================================================= */
const canvas = document.getElementById("c");
const ctx = canvas.getContext("2d");

function resizeCanvas() {
  const dpr = window.devicePixelRatio || 1;
  canvas.width  = Math.floor(window.innerWidth  * dpr);
  canvas.height = Math.floor(window.innerHeight * dpr);
  canvas.style.width  = window.innerWidth  + "px";
  canvas.style.height = window.innerHeight + "px";
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
}
window.addEventListener("resize", resizeCanvas);
resizeCanvas();

/* =========================================================
   INPUT DE TECLADO
   ========================================================= */
const keys = Object.create(null);

window.addEventListener("keydown", (e) => {
  const k = e.key.toLowerCase();
  keys[k] = true;
  if (e.key === " " || e.key.startsWith("Arrow")) e.preventDefault();

  // Selector de tipo de celda con 1-6
  if (k >= "1" && k <= "6") {
    const idx = parseInt(k, 10) - 1;
    tipoSeleccionado = TECLAS_TIPO[idx];
  }
});
window.addEventListener("keyup", (e) => {
  keys[e.key.toLowerCase()] = false;
});
window.addEventListener("blur", () => {
  for (const k in keys) keys[k] = false;
});

/* =========================================================
   INPUT DE RATÓN (construcción por arrastre)
   ========================================================= */
let tipoSeleccionado = "muro";
let mouseDown = false;
let lastPaintCol = -1;
let lastPaintRow = -1;

// Convierte coordenadas de pantalla → celda del grid (o null si fuera)
function pantallaACelda(px, py) {
  const off = getGridOffset();
  const gx = px - off.x;
  const gy = py - off.y;
  if (gx < 0 || gy < 0 || gx >= GRID_W || gy >= GRID_H) return null;
  return {
    col: Math.floor(gx / TILE),
    row: Math.floor(gy / TILE)
  };
}

/** Pinta una celda (y notifica al render para que redibuje su caché). */
function pintarCelda(col, row, tipo) {
  if (col < 0 || row < 0 || col >= COLS || row >= ROWS) return;
  const celda = grid[row][col];
  if (celda.tipo === tipo) return; // sin cambios reales
  celda.tipo = tipo;
  gridDirty = true;
}

canvas.addEventListener("mousedown", (e) => {
  if (e.button !== 0) return; // solo botón izquierdo
  const celda = pantallaACelda(e.clientX, e.clientY);
  if (!celda) return;
  mouseDown = true;
  pintarCelda(celda.col, celda.row, tipoSeleccionado);
  lastPaintCol = celda.col;
  lastPaintRow = celda.row;
});

canvas.addEventListener("mousemove", (e) => {
  if (!mouseDown) return;
  const celda = pantallaACelda(e.clientX, e.clientY);
  if (!celda) return;
  // Pintamos línea entre la última celda y la actual para no dejar huecos
  // al arrastrar rápido con el ratón.
  pintarLinea(lastPaintCol, lastPaintRow, celda.col, celda.row);
  lastPaintCol = celda.col;
  lastPaintRow = celda.row;
});

// Terminar arrastre en cualquier sitio
window.addEventListener("mouseup", () => {
  mouseDown = false;
  lastPaintCol = -1;
  lastPaintRow = -1;
});

/** Algoritmo de Bresenham simplificado para pintar línea de celdas. */
function pintarLinea(c0, r0, c1, r1) {
  if (c0 < 0 || r0 < 0) {
    pintarCelda(c1, r1, tipoSeleccionado);
    return;
  }
  let dc = Math.abs(c1 - c0);
  let dr = Math.abs(r1 - r0);
  let sc = c0 < c1 ? 1 : -1;
  let sr = r0 < r1 ? 1 : -1;
  let err = dc - dr;
  let c = c0, r = r0;
  while (true) {
    pintarCelda(c, r, tipoSeleccionado);
    if (c === c1 && r === r1) break;
    const e2 = 2 * err;
    if (e2 > -dr) { err -= dr; c += sc; }
    if (e2 <  dc) { err += dc; r += sr; }
  }
}

/* =========================================================
   OFFSET DEL GRID (centrado en pantalla)
   ========================================================= */
function getGridOffset() {
  return {
    x: Math.floor((window.innerWidth  - GRID_W) / 2),
    y: Math.floor((window.innerHeight - GRID_H) / 2)
  };
}

/* =========================================================
   GRID DE CONSTRUCCIÓN
   ========================================================= */
const grid = [];
for (let r = 0; r < ROWS; r++) {
  const fila = [];
  for (let c = 0; c < COLS; c++) {
    fila.push({ tipo: "vacio" });
  }
  grid.push(fila);
}

// Flag para saber si hay que redibujar la caché del grid
let gridDirty = true;

// Canvas offscreen donde cacheamos la capa del grid (celdas + líneas).
// Solo se redibuja cuando gridDirty = true.
const gridCanvas = document.createElement("canvas");
gridCanvas.width  = GRID_W;
gridCanvas.height = GRID_H;
const gridCtx = gridCanvas.getContext("2d");

/** Consulta si una celda es transitable (no muro y dentro del grid). */
function celdaEsTransitable(col, row) {
  if (col < 0 || row < 0 || col >= COLS || row >= ROWS) return false;
  return grid[row][col].tipo !== "muro";
}

/** Devuelve true si (x,y) en píxeles de mundo cae en un muro. */
function puntoEnMuro(x, y) {
  const col = Math.floor(x / TILE);
  const row = Math.floor(y / TILE);
  if (col < 0 || row < 0 || col >= COLS || row >= ROWS) return true;
  return grid[row][col].tipo === "muro";
}

/** Redibuja la caché del grid (celdas + líneas). */
function redibujarGridCache() {
  gridCtx.clearRect(0, 0, GRID_W, GRID_H);

  // Fondo base del área del grid
  gridCtx.fillStyle = "#151526";
  gridCtx.fillRect(0, 0, GRID_W, GRID_H);

  // Relleno de celdas no vacías
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      const tipo = grid[r][c].tipo;
      const color = TIPOS_CELDA[tipo].color;
      if (color) {
        gridCtx.fillStyle = color;
        gridCtx.fillRect(c * TILE, r * TILE, TILE, TILE);
      }
    }
  }

  // Líneas del grid
  gridCtx.strokeStyle = COLOR_GRID;
  gridCtx.lineWidth = 1;
  gridCtx.beginPath();
  for (let x = 0; x <= COLS; x++) {
    const px = x * TILE + 0.5;
    gridCtx.moveTo(px, 0);
    gridCtx.lineTo(px, GRID_H);
  }
  for (let y = 0; y <= ROWS; y++) {
    const py = y * TILE + 0.5;
    gridCtx.moveTo(0, py);
    gridCtx.lineTo(GRID_W, py);
  }
  gridCtx.stroke();

  // Borde exterior
  gridCtx.strokeStyle = "rgba(255,255,255,0.18)";
  gridCtx.strokeRect(0.5, 0.5, GRID_W, GRID_H);

  gridDirty = false;
}

/* =========================================================
   SISTEMA DE NPCs
   ========================================================= */
let nextNPCId = 1;

function crearNPC(tipo, x, y) {
  return {
    id: nextNPCId++,
    tipo,
    x, y,
    angle: Math.random() * Math.PI * 2,
    speed: NPC_SPEEDS[tipo],
    targetX: x,
    targetY: y,
    color: NPC_COLORS[tipo]
  };
}

/** Busca una celda transitable aleatoria y devuelve su centro en píxeles. */
function elegirTargetAleatorio(npc) {
  // Intentos acotados: si el grid está muy lleno de muros, no colgarse
  for (let i = 0; i < 50; i++) {
    const col = Math.floor(Math.random() * COLS);
    const row = Math.floor(Math.random() * ROWS);
    if (celdaEsTransitable(col, row)) {
      npc.targetX = col * TILE + TILE / 2;
      npc.targetY = row * TILE + TILE / 2;
      return;
    }
  }
  // Fallback: quedarse en el sitio
  npc.targetX = npc.x;
  npc.targetY = npc.y;
}

function lerpAngle(a, b, t) {
  let diff = ((b - a + Math.PI * 3) % (Math.PI * 2)) - Math.PI;
  return a + diff * t;
}

/** Actualiza posición y ángulo del NPC. Si choca con muro, nuevo target. */
function actualizarNPC(npc, dt) {
  const dx = npc.targetX - npc.x;
  const dy = npc.targetY - npc.y;
  const dist = Math.hypot(dx, dy);

  if (dist < 2) {
    elegirTargetAleatorio(npc);
    return;
  }

  const dirDeseada = Math.atan2(dy, dx);
  const t = Math.min(1, dt * 8);
  npc.angle = lerpAngle(npc.angle, dirDeseada, t);

  const step = npc.speed * dt;
  // Calculamos el siguiente paso ANTES de mover
  const nx = npc.x + Math.cos(dirDeseada) * Math.min(step, dist);
  const ny = npc.y + Math.sin(dirDeseada) * Math.min(step, dist);

  // Comprobación anti-muro en el punto destino
  if (puntoEnMuro(nx, ny)) {
    // Choca: elegimos un nuevo target transitable
    elegirTargetAleatorio(npc);
    return;
  }

  npc.x = nx;
  npc.y = ny;
}

/* =========================================================
   DIBUJO DE NPCs
   ========================================================= */
function hexToRgba(hex, alpha) {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return `rgba(${r},${g},${b},${alpha})`;
}

function dibujarNPC(ctx, npc, offsetX, offsetY) {
  const sx = npc.x + offsetX;
  const sy = npc.y + offsetY;

  const a1 = npc.angle - VISION_FOV / 2;
  const a2 = npc.angle + VISION_FOV / 2;
  ctx.beginPath();
  ctx.moveTo(sx, sy);
  ctx.lineTo(sx + Math.cos(a1) * VISION_RADIUS,
             sy + Math.sin(a1) * VISION_RADIUS);
  ctx.lineTo(sx + Math.cos(a2) * VISION_RADIUS,
             sy + Math.sin(a2) * VISION_RADIUS);
  ctx.closePath();
  ctx.fillStyle = hexToRgba(npc.color, VISION_ALPHA);
  ctx.fill();

  ctx.beginPath();
  ctx.arc(sx, sy, NPC_RADIUS, 0, Math.PI * 2);
  ctx.fillStyle = npc.color;
  ctx.fill();
  ctx.lineWidth = 2;
  ctx.strokeStyle = "#ffffff";
  ctx.stroke();
}

/* =========================================================
   POBLAR EL MUNDO
   ========================================================= */
const npcs = [];

function posAleatoriaTransitable() {
  for (let i = 0; i < 50; i++) {
    const col = Math.floor(Math.random() * COLS);
    const row = Math.floor(Math.random() * ROWS);
    if (celdaEsTransitable(col, row)) {
      return {
        x: col * TILE + TILE / 2,
        y: row * TILE + TILE / 2
      };
    }
  }
  return { x: TILE / 2, y: TILE / 2 };
}

function poblarMundo() {
  const spawn = (tipo, n) => {
    for (let i = 0; i < n; i++) {
      const p = posAleatoriaTransitable();
      const npc = crearNPC(tipo, p.x, p.y);
      elegirTargetAleatorio(npc);
      npcs.push(npc);
    }
  };
  spawn("preso", 5);
  spawn("guardia", 2);
  spawn("obrero", 1);
}
poblarMundo();

/* =========================================================
   SISTEMA DE TICKS
   ========================================================= */
let lastTick = performance.now();
function tick() {
  // Reservado para lógica de simulación lenta
}

/* =========================================================
   FPS
   ========================================================= */
let fpsAccum = 0;
let fpsFrames = 0;
let fpsValue = 0;

/* =========================================================
   DIBUJO GENERAL
   ========================================================= */
function drawFPS() {
  ctx.font = "14px monospace";
  ctx.textBaseline = "top";
  ctx.fillStyle = "#e0e0e0";
  ctx.fillText(`FPS: ${fpsValue}`, 10, 10);
  ctx.fillText(`NPCs: ${npcs.length}`, 10, 28);
}

function drawTipoSeleccionado() {
  const txt = `Tipo: ${tipoSeleccionado}`;
  ctx.font = "14px monospace";
  ctx.textBaseline = "top";
  // Fondo del panelito
  const pad = 6;
  const w = ctx.measureText(txt).width + pad * 2;
  const h = 20;
  const x = window.innerWidth - w - 10;
  const y = 10;
  ctx.fillStyle = "rgba(0,0,0,0.55)";
  ctx.fillRect(x, y, w, h);
  // Muestreo del color del tipo seleccionado
  const col = TIPOS_CELDA[tipoSeleccionado].color;
  if (col) {
    ctx.fillStyle = col;
    ctx.fillRect(x + 3, y + 4, 12, h - 8);
  }
  // Texto
  ctx.fillStyle = "#e0e0e0";
  ctx.fillText(txt, x + (col ? 20 : pad), y + 3);
}

function draw() {
  const off = getGridOffset();

  // Fondo global
  ctx.fillStyle = COLOR_BG;
  ctx.fillRect(0, 0, window.innerWidth, window.innerHeight);

  // Redibujamos la caché del grid solo si cambió
  if (gridDirty) redibujarGridCache();

  // Pegamos la caché del grid centrada
  ctx.drawImage(gridCanvas, off.x, off.y);

  // NPCs
  for (const npc of npcs) {
    dibujarNPC(ctx, npc, off.x, off.y);
  }

  drawFPS();
  drawTipoSeleccionado();
}

/* =========================================================
   BUCLE PRINCIPAL
   ========================================================= */
let lastFrameTime = performance.now();

function loop(now) {
  const dtMs = now - lastFrameTime;
  lastFrameTime = now;
  const dt = Math.min(dtMs / 1000, 0.05);

  // --- Ticks de simulación ---
  let ticksRun = 0;
  while (now - lastTick >= TICK_MS && ticksRun < 5) {
    lastTick += TICK_MS;
    tick();
    ticksRun++;
  }
  if (now - lastTick >= TICK_MS) lastTick = now;

  // --- Actualización de NPCs ---
  for (const npc of npcs) actualizarNPC(npc, dt);

  // --- FPS ---
  fpsAccum += dtMs;
  fpsFrames++;
  if (fpsAccum >= 500) {
    fpsValue = Math.round((fpsFrames * 1000) / fpsAccum);
    fpsAccum = 0;
    fpsFrames = 0;
  }

  // --- Render ---
  draw();

  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
